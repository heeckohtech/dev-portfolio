debugChallenge:
  brokenCode: |
    fetch("/api/user")
      .then(res => res.json)
      .then(data => console.log(data));
  hints:
    - "Look closely at the second .then() — what type is res.json without parentheses?"
    - "res.json is a method, not a value. Calling it without () passes the function itself, not its result."
    - "You need res.json() — calling it — since it returns a Promise that resolves to the parsed data."
  solution: |
    fetch("/api/user")
      .then(res => res.json())
      .then(data => console.log(data));
  solutionExplanation: "res.json is a method that must be called to actually parse the response body. Without the parentheses, you're passing the function reference itself into the next .then(), not the parsed data."
